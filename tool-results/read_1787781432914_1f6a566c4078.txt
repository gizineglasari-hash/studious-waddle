    83→function simpleHash(password: string): string {
    84→  // This is NOT cryptographically secure.
    85→  // For production, use Supabase Auth or NextAuth with proper hashing.
    86→  let hash = 0;
    87→  const salt = "GEMAS_2026_SALT_KEY";
    88→  const str = password + salt;
    89→  for (let i = 0; i < str.length; i++) {
    90→    const char = str.charCodeAt(i);
    91→    hash = (hash << 5) - hash + char;
    92→    hash = hash & hash;
    93→  }
    94→  // Convert to hex and add extra layer
    95→  const hex = Math.abs(hash).toString(16).padStart(8, "0");
    96→  return btoa(hex + salt + password.length);
    97→}

    99→function generateId(): string {
   100→  return `${Date.now()}-${Math.random().toString(36).substr(2, 9)}`;
   101→}

   103→// Generate valid UUID for Supabase tables (all id columns are UUID type)
   104→function generateUUID(): string {
   105→  if (typeof crypto !== "undefined" && typeof crypto.randomUUID === "function") {
   106→    return crypto.randomUUID();
   107→  }
   108→  // Fallback for older browsers
   109→  return "xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx".replace(/[xy]/g, function (c) {
   110→    const r = (Math.random() * 16) | 0;
   111→    const v = c === "x" ? r : (r & 0x3) | 0x8;
   112→    return v.toString(16);
   113→  });
   114→}
   115→
   116→// Pick the right ID generator based on Supabase availability
   117→function genId(): string {
   118→  return isSupabaseConfigured ? generateUUID() : generateId();
   119→}

   360→// =====================================================
   361→// STORE IMPLEMENTATION
   362→//
   363→// KEY DESIGN PRINCIPLE (per spec rule #6):
   364→//   "Use async where needed but keep the store interface synchronous
   365→//    (use internal async, return sync)."
   366→//
   367→// All public method signatures stay EXACTLY the same as before (sync return
   368→// types) so existing callers don't need any changes. When Supabase is
   369→// configured, methods fire async Supabase operations in the background
   370→// (fire-and-forget IIFE) and apply optimistic updates to local state
   371→// immediately. When Supabase is NOT configured, the original localStorage
   372→// logic runs unchanged.
   373→//
   374→// For `login()`/`adminLogin()` specifically, we set the current user/admin
   375→// id optimistically (to a placeholder id) so the dashboard renders right
   376→// away. When the async Supabase response arrives, we replace the
   377→// placeholder with the real user id and call `refreshData()` to populate
   378→// consultations/notifications/children from the server.
   379→// =====================================================
   380→
   381→export const useAuthStore = create<AuthState>()(
   382→  persist(
   383→    (set, get) => ({
   384→      currentUserId: null,
   385→      currentAdminId: null,
   386→      isAdmin: false,
   387→      isAuthLoading: false,
   388→      authError: null,
   389→      lastRefreshedAt: null,
   390→      users: [],
   391→      children: [],
   392→      consultations: [],
   393→      notifications: [],
   394→
   395→      // ============ AUTH ============
   396→      register: (data) => {
   397→        const emailLower = data.email.toLowerCase().trim();
   398→
   399→        // Sync validation
   400→        if (!data.namaOrangTua.trim()) {
   401→          return { success: false, message: "Nama orang tua wajib diisi." };
   402→        }
   403→        if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(emailLower)) {
   404→          return { success: false, message: "Format email tidak valid." };
   405→        }
   406→        if (data.password.length < 6) {
   407→          return { success: false, message: "Password minimal 6 karakter." };
   408→        }
   409→        if (!data.nomorTelepon.trim()) {
   410→          return { success: false, message: "Nomor telepon wajib diisi." };
   411→        }
   412→
   413→        if (isSupabaseConfigured && supabase) {
   414→          // ALSO save to localStorage so user can login immediately
   415→          // (even if Supabase email confirmation is pending)
   416→          const now = new Date().toISOString();
   417→          const localUser: UserProfile = {
   418→            id: genId(),
   419→            namaOrangTua: data.namaOrangTua.trim(),
   420→            email: emailLower,
   421→            nomorTelepon: data.nomorTelepon.trim(),
   422→            alamat: data.alamat.trim(),
   423→            passwordHash: simpleHash(data.password),
   424→            role: "user",
   425→            createdAt: now,
   426→            updatedAt: now,
   427→          };
   428→          set((state) => ({ users: [...state.users, localUser] }));
   429→
   430→          // Fire signUp to Supabase in background
   431→          void (async () => {
   432→            try {
   433→              const { data: signUpData, error } = await supabase.auth.signUp({
   434→                email: emailLower,
   435→                password: data.password,
   436→                options: {
   437→                  data: {
   438→                    nama_orang_tua: data.namaOrangTua.trim(),
   439→                    nomor_telepon: data.nomorTelepon.trim(),
   440→                    alamat: data.alamat.trim(),
   441→                  },
   442→                },
   443→              });
   444→
   445→              if (error) {
   446→                console.error("[Supabase register] error:", error.message);
   447→                return;
   448→              }
   449→
   450→              // Cache the new profile locally so the user can log in
   451→              // immediately after registration.
   452→              if (signUpData.user) {
   453→                const now = new Date().toISOString();
   454→                const cachedProfile: UserProfile = {
   455→                  id: signUpData.user.id,
   456→                  namaOrangTua: data.namaOrangTua.trim(),
   457→                  email: emailLower,
   458→                  nomorTelepon: data.nomorTelepon.trim(),
   459→                  alamat: data.alamat.trim(),
   460→                  passwordHash: "",
   461→                  role: "user",
   462→                  createdAt: signUpData.user.created_at || now,
   463→                  updatedAt: now,
   464→                };
   465→                set((state) => ({
   466→                  users: [
   467→                    ...state.users.filter((u) => u.id !== cachedProfile.id),
   468→                    cachedProfile,
   469→                  ],
   470→                }));
   471→              }
   472→            } catch (err) {
   473→              console.error("[Supabase register] exception:", err);
   474→            }
   475→          })();
   476→
   477→          return { success: true, message: "Registrasi berhasil. Silakan login." };
   478→        }
   479→
   480→        // ----- localStorage fallback -----
   481→        const { users } = get();
   482→        if (users.some((u) => u.email.toLowerCase() === emailLower)) {
   483→          return { success: false, message: "Email sudah terdaftar. Silakan login." };
   484→        }
   485→
   486→        const newUser: UserProfile = {
   487→          id: genId(),
   488→          namaOrangTua: data.namaOrangTua.trim(),
   489→          email: emailLower,
   490→          nomorTelepon: data.nomorTelepon.trim(),
   491→          alamat: data.alamat.trim(),
   492→          passwordHash: simpleHash(data.password),
   493→          role: "user",
   494→          createdAt: new Date().toISOString(),
   495→          updatedAt: new Date().toISOString(),
   496→        };
   497→
   498→        set((state) => ({ users: [...state.users, newUser] }));
   499→        return { success: true, message: "Registrasi berhasil. Silakan login." };
   500→      },
   501→
   502→      login: (email, password) => {
   503→        const emailLower = email.toLowerCase().trim();
   504→
   505→        if (!emailLower) {
   506→          return { success: false, message: "Email wajib diisi." };
   507→        }
   508→        if (!password) {
   509→          return { success: false, message: "Password wajib diisi." };
   510→        }
   511→
   512→        if (isSupabaseConfigured && supabase) {
   513→          // For Supabase mode: use SYNCHRONOUS fallback approach
   514→          // Try to sign in synchronously using a workaround:
   515→          // Set loading state, then let the async complete
   516→          // BUT return a special message that tells the UI to wait
   517→
   518→          // Actually, we can't make this truly synchronous.
   519→          // Best approach: use localStorage fallback for immediate login,
   520→          // then sync with Supabase in background.
   521→
   522→          // Check if user exists in localStorage (from previous registration)
   523→          const { users } = get();
   524→          const localUser = users.find(
   525→            (u) => u.email.toLowerCase() === emailLower && u.role === "user"
   526→          );
   527→
   528→          if (localUser && localUser.passwordHash === simpleHash(password)) {
   529→            // Found in localStorage - login immediately
   530→            set({
   531→              currentUserId: localUser.id,
   532→              isAdmin: false,
   533→              isAuthLoading: false,
   534→              authError: null,
   535→            });
   536→
   537→            // Sync with Supabase in background
   538→            void (async () => {
   539→              try {
   540→                const { data: signInData, error } =
   541→                  await supabase.auth.signInWithPassword({
   542→                    email: emailLower,
   543→                    password,
   544→                  });
   545→
   546→                if (error) {
   547→                  console.warn("[Supabase login sync] error:", error.message);
   548→                  // Don't clear local login - user is already in
   549→                  return;
   550→                }
   551→
   552→                if (signInData.user) {
   553→                  // Update with Supabase UUID
   554→                  set({ currentUserId: signInData.user.id });
   555→                  await get().refreshData();
   556→                }
   557→              } catch (err) {
   558→                console.warn("[Supabase login sync] exception:", err);
   559→              }
   560→            })();
   561→
   562→            return { success: true, message: "Login berhasil." };
   563→          }
   564→
   565→          // Not found in localStorage - try Supabase directly
   566→          // We need to return synchronously, so we'll use a different approach:
   567→          // Set loading state and return a "pending" message
   568→          set({ isAuthLoading: true, authError: null });
   569→
   570→          // Fire async login
   571→          void (async () => {
   572→            try {
   573→              const { data: signInData, error } =
   574→                await supabase.auth.signInWithPassword({
   575→                  email: emailLower,
   576→                  password,
   577→                });
   578→
   579→              if (error || !signInData.user) {
   580→                console.error("[Supabase login] error:", error?.message);
   581→                set({
   582→                  currentUserId: null,
   583→                  isAuthLoading: false,
   584→                  authError: error?.message || "Login gagal.",
   585→                });
   586→                return;
   587→              }
   588→
   589→              const userId = signInData.user.id;
   590→
   591→              // Fetch profile
   592→              let profile: UserProfile | null = null;
   593→              try {
   594→                const { data: profileRow } = await supabase
   595→                  .from("profiles")
   596→                  .select("*")
   597→                  .eq("id", userId)
   598→                  .maybeSingle();
   599→
   600→                if (profileRow) {
   601→                  profile = mapProfileRow(profileRow as ProfileRow, signInData.user.email);
   602→                } else {
