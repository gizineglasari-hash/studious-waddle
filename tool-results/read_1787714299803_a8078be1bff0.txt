    63→    63→function getBrowserInfo(): Partial<RealTimeInfo> {
    64→    64→  if (typeof navigator === "undefined") return {};
    65→    65→  const ua = navigator.userAgent;
    66→    66→  let browser = "Unknown";
    67→    67→  let os = "Unknown";
    68→    68→  let device = "Desktop";
    69→    69→
    70→    70→  // Detect browser
    71→    71→  if (ua.includes("Edg/")) browser = "Microsoft Edge";
    72→    72→  else if (ua.includes("Chrome/")) browser = "Google Chrome";
    73→    73→  else if (ua.includes("Firefox/")) browser = "Mozilla Firefox";
    74→    74→  else if (ua.includes("Safari/")) browser = "Apple Safari";
    75→    75→  else if (ua.includes("Opera/") || ua.includes("OPR/")) browser = "Opera";
    76→    76→
    77→    77→  // Detect OS
    78→    78→  if (ua.includes("Windows")) os = "Windows";
    79→    79→  else if (ua.includes("Mac OS")) os = "macOS";
    80→    80→  else if (ua.includes("Android")) os = "Android";
    81→    81→  else if (ua.includes("iOS") || ua.includes("iPhone") || ua.includes("iPad")) os = "iOS";
    82→    82→  else if (ua.includes("Linux")) os = "Linux";
    83→    83→
    84→    84→  // Detect device
    85→    85→  if (/Mobi|Android/i.test(ua)) device = "Mobile";
    86→    86→  else if (/iPad|Tablet/i.test(ua)) device = "Tablet";
    87→    87→
    88→    88→  return {
    89→    89→    browser,
    90→    90→    os,
    91→    91→    device,
    92→    92→    screenSize: `${window.screen.width}x${window.screen.height}`,
    93→    93→    language: navigator.language,
    94→    94→  };
    95→    95→}

    97→    97→async function fetchIpInfo(): Promise<Partial<RealTimeInfo>> {
    98→    98→  try {
    99→    99→    const response = await fetch("https://ipapi.co/json/");
   100→   100→    if (response.ok) {
   101→   101→      const data = await response.json();
   102→   102→      return {
   103→   103→        ip: data.ip || "Unknown",
   104→   104→        city: data.city || "Unknown",
   105→   105→        region: data.region || "Unknown",
   106→   106→        country: data.country_name || "Unknown",
   107→   107→        timezone: data.timezone || "Unknown",
   108→   108→        isp: data.org || "Unknown",
   109→   109→      };
   110→   110→    }
   111→   111→  } catch (e) {
   112→   112→    // Fallback if API fails
   113→   113→  }
   114→   114→  return { ip: "Unknown", city: "Unknown", country: "Unknown" };
   115→   115→}

   117→   117→function loadVisitStats(): VisitStats {
   118→   118→  if (typeof localStorage === "undefined") {

   121→   121→      uniqueDays: new Set(),

   126→   126→      visitsByDevice: {},
   127→   127→      visitHistory: [],
   128→   128→    };
   129→   129→  }
   130→   130→  try {
   131→   131→    const stored = localStorage.getItem("gemas-analytics-stats");
   132→   132→    if (stored) {
   133→   133→      const data = JSON.parse(stored);
   134→   134→      return {
   135→   135→        totalVisits: data.totalVisits || 0,
   136→   136→        uniqueDays: new Set(data.uniqueDays || []),
   137→   137→        firstVisit: data.firstVisit || "",
   138→   138→        lastVisit: data.lastVisit || "",
   139→   139→        visitsByPage: data.visitsByPage || {},
   140→   140→        visitsByBrowser: data.visitsByBrowser || {},
   141→   141→        visitsByDevice: data.visitsByDevice || {},
   142→   142→        visitHistory: data.visitHistory || [],
   143→   143→      };
   144→   144→    }
   145→   145→  } catch (e) {
   146→   146→    // ignore parse errors
   147→   147→  }
   148→   148→  return {
   149→   149→    totalVisits: 0,
   150→   150→    uniqueDays: new Set(),
   151→   151→    firstVisit: "",
   152→   152→    lastVisit: "",
   153→   153→    visitsByPage: {},
   154→   154→    visitsByBrowser: {},
   155→   155→    visitsByDevice: {},
   156→   156→    visitHistory: [],
   157→   157→  };
   158→   158→}

   160→   160→function saveVisitStats(stats: VisitStats) {
   161→   161→  if (typeof localStorage === "undefined") return;
   162→   162→  try {
   163→   163→    const data = {
   164→   164→      totalVisits: stats.totalVisits,
   165→   165→      uniqueDays: Array.from(stats.uniqueDays),
   166→   166→      firstVisit: stats.firstVisit,
   167→   167→      lastVisit: stats.lastVisit,
   168→   168→      visitsByPage: stats.visitsByPage,
   169→   169→      visitsByBrowser: stats.visitsByBrowser,
   170→   170→      visitsByDevice: stats.visitsByDevice,
   171→   171→      visitHistory: stats.visitHistory.slice(-50), // keep last 50
   172→   172→    };
   173→   173→    localStorage.setItem("gemas-analytics-stats", JSON.stringify(data));
   174→   174→  } catch (e) {
   175→   175→    // ignore
   176→   176→  }
   177→   177→}
