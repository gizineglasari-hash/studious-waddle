    83→    83→function simpleHash(password: string): string {
    84→    84→  // This is NOT cryptographically secure.
    85→    85→  // For production, use Supabase Auth or NextAuth with proper hashing.
    86→    86→  let hash = 0;
    87→    87→  const salt = "GEMAS_2026_SALT_KEY";
    88→    88→  const str = password + salt;
    89→    89→  for (let i = 0; i < str.length; i++) {
    90→    90→    const char = str.charCodeAt(i);
    91→    91→    hash = (hash << 5) - hash + char;
    92→    92→    hash = hash & hash;
    93→    93→  }
    94→    94→  // Convert to hex and add extra layer
    95→    95→  const hex = Math.abs(hash).toString(16).padStart(8, "0");
    96→    96→  return btoa(hex + salt + password.length);
    97→    97→}

   103→   103→// Generate valid UUID for Supabase tables (all id columns are UUID type)
   104→   104→function generateUUID(): string {
   105→   105→  if (typeof crypto !== "undefined" && typeof crypto.randomUUID === "function") {
   106→   106→    return crypto.randomUUID();
   107→   107→  }
   108→   108→  // Fallback for older browsers
   109→   109→  return "xxxxxxxx-xxxx-4xxx-yxxx-xxxxxxxxxxxx".replace(/[xy]/g, function (c) {
   110→   110→    const r = (Math.random() * 16) | 0;
   111→   111→    const v = c === "x" ? r : (r & 0x3) | 0x8;
   112→   112→    return v.toString(16);
   113→   113→  });
   114→   114→}

   360→   360→// =====================================================
   361→   361→// STORE IMPLEMENTATION
   362→   362→//
   363→   363→// KEY DESIGN PRINCIPLE (per spec rule #6):
   364→   364→//   "Use async where needed but keep the store interface synchronous
   365→   365→//    (use internal async, return sync)."
   366→   366→//
   367→   367→// All public method signatures stay EXACTLY the same as before (sync return
   368→   368→// types) so existing callers don't need any changes. When Supabase is
   369→   369→// configured, methods fire async Supabase operations in the background
   370→   370→// (fire-and-forget IIFE) and apply optimistic updates to local state
   371→   371→// immediately. When Supabase is NOT configured, the original localStorage
   372→   372→// logic runs unchanged.
   373→   373→//
   374→   374→// For `login()`/`adminLogin()` specifically, we set the current user/admin
   375→   375→// id optimistically (to a placeholder id) so the dashboard renders right
   376→   376→// away. When the async Supabase response arrives, we replace the
   377→   377→// placeholder with the real user id and call `refreshData()` to populate
   378→   378→// consultations/notifications/children from the server.
   379→   379→// =====================================================
   380→   380→
   381→   381→export const useAuthStore = create<AuthState>()(
   382→   382→  persist(
   383→   383→    (set, get) => ({
   384→   384→      currentUserId: null,
   385→   385→      currentAdminId: null,
   386→   386→      isAdmin: false,
   387→   387→      isAuthLoading: false,
   388→   388→      authError: null,
   389→   389→      lastRefreshedAt: null,
   390→   390→      users: [],
   391→   391→      children: [],
   392→   392→      consultations: [],
   393→   393→      notifications: [],
   394→   394→
   395→   395→      // ============ AUTH ============
   396→   396→      register: (data) => {
   397→   397→        const emailLower = data.email.toLowerCase().trim();
   398→   398→
   399→   399→        // Sync validation
   400→   400→        if (!data.namaOrangTua.trim()) {
   401→   401→          return { success: false, message: "Nama orang tua wajib diisi." };
   402→   402→        }
   403→   403→        if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(emailLower)) {
   404→   404→          return { success: false, message: "Format email tidak valid." };
   405→   405→        }
   406→   406→        if (data.password.length < 6) {
   407→   407→          return { success: false, message: "Password minimal 6 karakter." };
   408→   408→        }
   409→   409→        if (!data.nomorTelepon.trim()) {
   410→   410→          return { success: false, message: "Nomor telepon wajib diisi." };
   411→   411→        }
   412→   412→
   413→   413→        if (isSupabaseConfigured && supabase) {
   414→   414→          // ALSO save to localStorage so user can login immediately
   415→   415→          // (even if Supabase email confirmation is pending)
   416→   416→          const now = new Date().toISOString();
   417→   417→          const localUser: UserProfile = {
   418→   418→            id: genId(),
   419→   419→            namaOrangTua: data.namaOrangTua.trim(),
   420→   420→            email: emailLower,
   421→   421→            nomorTelepon: data.nomorTelepon.trim(),
   422→   422→            alamat: data.alamat.trim(),
   423→   423→            passwordHash: simpleHash(data.password),
   424→   424→            role: "user",
   425→   425→            createdAt: now,
   426→   426→            updatedAt: now,
   427→   427→          };
   428→   428→          set((state) => ({ users: [...state.users, localUser] }));
   429→   429→
   430→   430→          // Fire signUp to Supabase in background
   431→   431→          void (async () => {
   432→   432→            try {
   433→   433→              const { data: signUpData, error } = await supabase.auth.signUp({
   434→   434→                email: emailLower,
   435→   435→                password: data.password,
   436→   436→                options: {
   437→   437→                  data: {
   438→   438→                    nama_orang_tua: data.namaOrangTua.trim(),
   439→   439→                    nomor_telepon: data.nomorTelepon.trim(),
   440→   440→                    alamat: data.alamat.trim(),
   441→   441→                  },
   442→   442→                },
   443→   443→              });
   444→   444→
   445→   445→              if (error) {
   446→   446→                console.error("[Supabase register] error:", error.message);
   447→   447→                return;
   448→   448→              }
   449→   449→
   450→   450→              // Cache the new profile locally so the user can log in
   451→   451→              // immediately after registration.
   452→   452→              if (signUpData.user) {
   453→   453→                const now = new Date().toISOString();
   454→   454→                const cachedProfile: UserProfile = {
   455→   455→                  id: signUpData.user.id,
   456→   456→                  namaOrangTua: data.namaOrangTua.trim(),
   457→   457→                  email: emailLower,
   458→   458→                  nomorTelepon: data.nomorTelepon.trim(),
   459→   459→                  alamat: data.alamat.trim(),
   460→   460→                  passwordHash: "",
   461→   461→                  role: "user",
   462→   462→                  createdAt: signUpData.user.created_at || now,
   463→   463→                  updatedAt: now,
   464→   464→                };
   465→   465→                set((state) => ({
   466→   466→                  users: [
   467→   467→                    ...state.users.filter((u) => u.id !== cachedProfile.id),
   468→   468→                    cachedProfile,
   469→   469→                  ],
   470→   470→                }));
   471→   471→              }
   472→   472→            } catch (err) {
   473→   473→              console.error("[Supabase register] exception:", err);
   474→   474→            }
   475→   475→          })();
   476→   476→
   477→   477→          return { success: true, message: "Registrasi berhasil. Silakan login." };
   478→   478→        }
   479→   479→
   480→   480→        // ----- localStorage fallback -----
   481→   481→        const { users } = get();
   482→   482→        if (users.some((u) => u.email.toLowerCase() === emailLower)) {
   483→   483→          return { success: false, message: "Email sudah terdaftar. Silakan login." };
   484→   484→        }
   485→   485→
   486→   486→        const newUser: UserProfile = {
   487→   487→          id: genId(),
