   143→function getContentTypeIcon(
   144→  type: ContentType
   145→): React.ComponentType<{ className?: string }> {
   146→  switch (type) {
   147→    case "video":
   148→      return FileVideo;
   149→    case "image":
   150→      return FileImage;
   151→    case "pdf":
   152→      return FileText;
   153→    case "article":
   154→      return Newspaper;
   155→    case "banner":
   156→      return LayoutGrid;
   157→    default:
   158→      return FileText;
   159→  }
   160→}
   161→
   162→function formatFileSize(bytes: number): string {
   163→  if (bytes < 1024) return `${bytes} B`;
   164→  if (bytes < 1024 * 1024) return `${(bytes / 1024).toFixed(1)} KB`;
   165→  return `${(bytes / (1024 * 1024)).toFixed(2)} MB`;
   166→}
   167→
   168→function fileToDataUrl(file: File): Promise<string> {
   169→  return new Promise((resolve, reject) => {
   170→    const reader = new FileReader();
   171→    reader.onload = () => resolve(reader.result as string);
   172→    reader.onerror = () => reject(new Error("FileReader error"));
   173→    reader.readAsDataURL(file);
   174→  });
   175→}
   176→
   177→function getMediaAccept(type: ContentType): string {
   178→  switch (type) {
   179→    case "video":
   180→      return "video/*";
   181→    case "image":
   182→      return "image/*";
   183→    case "pdf":
   184→      return ".pdf,application/pdf";
   185→    default:
   186→      return "";
   187→  }
   188→}
   189→
   190→function getMediaAcceptLabel(type: ContentType): string {
   191→  switch (type) {
   192→    case "video":
   193→      return "video (MP4, WebM, MOV)";
   194→    case "image":
   195→      return "gambar (JPG, PNG, WebP)";
   196→    case "pdf":
   197→      return "PDF";
   198→    default:
   199→      return "file";
   200→  }
   201→}

   207→export function EditWebsiteView() {

   209→  const { setView } = useGemasStore();

   211→  const getCurrentUser = useAuthStore((s) => s.getCurrentUser);
   212→

   214→  const addContent = useContentStore((s) => s.addContent);
   215→  const updateContent = useContentStore((s) => s.updateContent);
   216→  const deleteContent = useContentStore((s) => s.deleteContent);
   217→  const toggleActive = useContentStore((s) => s.toggleActive);
   218→  const updateOrder = useContentStore((s) => s.updateOrder);
   219→
   220→  // Hydration-safe mount flag
   221→  const [mounted, setMounted] = useState(false);
   222→  useEffect(() => {
   223→    setMounted(true);
   224→  }, []);
   225→
   226→  // Redirect non-admins (safe effect — no derived state sync)
   227→  useEffect(() => {
   228→    if (mounted && !isAdmin) {
   229→      toast({
   230→        title: "Akses ditolak",
   231→        description: "Silakan login sebagai admin terlebih dahulu.",
   232→        variant: "destructive",
   233→      });
   234→      setView("admin-login");
   235→    }
   236→  }, [mounted, isAdmin, setView, toast]);
   237→
   238→  // UI state
   239→  const [activeFilter, setActiveFilter] = useState<FilterTabKey>("all");
   240→  const [searchQuery, setSearchQuery] = useState("");
   241→  const [dialogOpen, setDialogOpen] = useState(false);
   242→  const [editingId, setEditingId] = useState<string | null>(null);
   243→  const [deleteTargetId, setDeleteTargetId] = useState<string | null>(null);
   244→  const [submitting, setSubmitting] = useState(false);
   245→  const [processingFile, setProcessingFile] = useState(false);
   246→  const [form, setForm] = useState<ContentFormData>(DEFAULT_FORM);
   247→
   248→  // Sorted (by displayOrder) — reactive to store changes
   249→  const sortedContents = useMemo(() => {
   250→    return [...contents].sort((a, b) => a.displayOrder - b.displayOrder);
   251→  }, [contents]);
   252→
   253→  // Filtered by tab + search
   254→  const filteredContents = useMemo(() => {
   255→    let list = sortedContents;
   256→    if (activeFilter !== "all") {
   257→      list = list.filter((c) => c.contentType === activeFilter);
   258→    }
   259→    const q = searchQuery.trim().toLowerCase();
   260→    if (q) {
   261→      list = list.filter((c) => c.title.toLowerCase().includes(q));
   262→    }
   263→    return list;
   264→  }, [sortedContents, activeFilter, searchQuery]);
   265→
   266→  // Counts per tab
   267→  const tabCounts = useMemo(() => {
   268→    const map: Record<FilterTabKey, number> = {
   269→      all: sortedContents.length,
   270→      video: 0,
   271→      image: 0,
   272→      pdf: 0,
   273→      article: 0,
   274→      banner: 0,
   275→    };
   276→    sortedContents.forEach((c) => {
   277→      map[c.contentType] = (map[c.contentType] ?? 0) + 1;
   278→    });
   279→    return map;
   280→  }, [sortedContents]);
   281→
   282→  // Available categories based on content type
   283→  const availableCategories = useMemo<ContentCategory[]>(() => {
   284→    return form.contentType === "video" ? VIDEO_CATEGORIES : CONTENT_CATEGORIES;
   285→  }, [form.contentType]);
   286→
   287→  // =====================================================
   288→  // HANDLERS
   289→  // =====================================================
   290→
   291→  const handleOpenAdd = () => {
   292→    const maxOrder =
   293→      sortedContents.length > 0
   294→        ? Math.max(...sortedContents.map((c) => c.displayOrder))
   295→        : 0;
   296→    setForm({ ...DEFAULT_FORM, displayOrder: maxOrder + 1 });
   297→    setEditingId(null);
   298→    setDialogOpen(true);
   299→  };
   300→
   301→  const handleOpenEdit = (content: EducationalContent) => {
   302→    setForm({
   303→      title: content.title,
   304→      description: content.description,
   305→      contentType: content.contentType,
   306→      category: content.category,
   307→      videoSource: content.videoSource ?? "upload",
   308→      externalUrl: content.externalUrl ?? "",
   309→      mediaUrl: content.mediaUrl ?? "",
   310→      thumbnailUrl: content.thumbnailUrl ?? "",
   311→      articleContent: content.articleContent ?? "",
   312→      duration: content.duration ?? "",
   313→      isActive: content.isActive,
   314→      displayOrder: content.displayOrder,
   315→      fileSize: content.fileSize ?? "",
   316→      mediaFileName: content.mediaUrl ? content.mediaUrl.split("/").pop() ?? "" : "",
   317→      thumbnailFileName: content.thumbnailUrl
   318→        ? content.thumbnailUrl.split("/").pop() ?? ""
   319→        : "",
   320→    });
   321→    setEditingId(content.id);
   322→    setDialogOpen(true);
   323→  };
   324→
   325→  const handleCloseDialog = () => {
   326→    setDialogOpen(false);
   327→    setEditingId(null);
   328→  };
   329→
   330→  const handleContentTypeChange = (type: ContentType) => {
   331→    setForm((prev) => {
   332→      const validCategories =
   333→        type === "video" ? VIDEO_CATEGORIES : CONTENT_CATEGORIES;
   334→      const categoryValid = validCategories.includes(prev.category);
   335→      return {
   336→        ...prev,
   337→        contentType: type,
   338→        category: categoryValid ? prev.category : validCategories[0],
   339→        // Reset media fields when changing type to avoid stale data
   340→        mediaUrl: "",
   341→        externalUrl: "",
   342→        mediaFileName: "",
   343→        fileSize: "",
   344→        videoSource: type === "video" ? prev.videoSource : "upload",
   345→      };
   346→    });
   347→  };
   348→
   349→  const handleVideoSourceChange = (source: VideoSource) => {
   350→    setForm((prev) => ({
   351→      ...prev,
   352→      videoSource: source,
   353→      // Clear media fields when switching source type
   354→      mediaUrl: "",
   355→      externalUrl: "",
   356→      mediaFileName: "",
   357→      fileSize: "",
   358→    }));
   359→  };
   360→
   361→  const handleMediaFileChange = async (
   362→    e: React.ChangeEvent<HTMLInputElement>
   363→  ) => {
   364→    const file = e.target.files?.[0];
   365→    if (!file) return;
   366→
   367→    if (file.size > MAX_FILE_SIZE_BYTES) {
   368→      toast({
   369→        title: "Peringatan ukuran file",
   370→        description:
   371→          "File lebih besar dari 5MB dapat melebihi batas penyimpanan browser (localStorage). Pertimbangkan untuk menggunakan URL eksternal sebagai gantinya.",
   372→        variant: "destructive",
   373→      });
   374→    }
   375→
   376→    setProcessingFile(true);
   377→    try {
   378→      const dataUrl = await fileToDataUrl(file);
   379→      setForm((prev) => ({
   380→        ...prev,
   381→        mediaUrl: dataUrl,
   382→        fileSize: formatFileSize(file.size),
   383→        mediaFileName: file.name,
   384→        externalUrl: "",
   385→      }));
   386→    } catch {
   387→      toast({
   388→        title: "Gagal memproses file",
   389→        description: "Terjadi kesalahan saat membaca file. Coba lagi.",
   390→        variant: "destructive",
   391→      });
   392→    } finally {
   393→      setProcessingFile(false);
   394→      // Reset input value so the same file can be re-selected
   395→      e.target.value = "";
   396→    }
   397→  };
   398→
   399→  const handleThumbnailFileChange = async (
   400→    e: React.ChangeEvent<HTMLInputElement>
   401→  ) => {
   402→    const file = e.target.files?.[0];
   403→    if (!file) return;
   404→
   405→    setProcessingFile(true);
   406→    try {
   407→      const dataUrl = await fileToDataUrl(file);
   408→      setForm((prev) => ({
   409→        ...prev,
   410→        thumbnailUrl: dataUrl,
   411→        thumbnailFileName: file.name,
   412→      }));
   413→    } catch {
   414→      toast({
   415→        title: "Gagal memproses thumbnail",
   416→        description: "Terjadi kesalahan saat membaca file thumbnail.",
   417→        variant: "destructive",
   418→      });
   419→    } finally {
   420→      setProcessingFile(false);
   421→      e.target.value = "";
   422→    }
   423→  };
   424→
   425→  const handleClearMedia = () => {
   426→    setForm((prev) => ({
   427→      ...prev,
   428→      mediaUrl: "",
   429→      fileSize: "",
   430→      mediaFileName: "",
   431→    }));
   432→  };
   433→
   434→  const handleClearThumbnail = () => {
   435→    setForm((prev) => ({
   436→      ...prev,
   437→      thumbnailUrl: "",
   438→      thumbnailFileName: "",
   439→    }));
   440→  };
   441→
   442→  const validateForm = (): string | null => {
   443→    if (!form.title.trim()) return "Judul wajib diisi.";
   444→    if (!form.description.trim()) return "Deskripsi wajib diisi.";
   445→
   446→    if (form.contentType === "video") {
   447→      if (form.videoSource === "upload" && !form.mediaUrl) {
   448→        return "File video wajib diunggah.";
   449→      }
   450→      if (form.videoSource !== "upload" && !form.externalUrl.trim()) {
   451→        return `URL video wajib diisi untuk sumber ${form.videoSource}.`;
   452→      }
   453→    }
   454→    if (form.contentType === "image" && !form.mediaUrl) {
   455→      return "Gambar wajib diunggah.";
   456→    }
   457→    if (form.contentType === "pdf" && !form.mediaUrl) {
   458→      return "File PDF wajib diunggah.";
   459→    }
   460→    return null;
   461→  };
   462→
   463→  const handleSubmit = () => {
   464→    const error = validateForm();
   465→    if (error) {
   466→      toast({ title: "Validasi gagal", description: error, variant: "destructive" });
   467→      return;
   468→    }
   469→
   470→    const adminEmail = getCurrentUser()?.email ?? "admin@gemas.id";
   471→
   472→    const payload = {
   473→      title: form.title.trim(),
   474→      description: form.description.trim(),
   475→      contentType: form.contentType,
   476→      category: form.category,
   477→      mediaUrl: form.mediaUrl || undefined,
   478→      externalUrl: form.externalUrl.trim() || undefined,
   479→      videoSource: form.contentType === "video" ? form.videoSource : undefined,
   480→      thumbnailUrl: form.thumbnailUrl || undefined,
   481→      articleContent:
   482→        form.contentType === "article" ? form.articleContent.trim() || undefined : undefined,
   483→      isActive: form.isActive,
   484→      displayOrder: form.displayOrder,
   485→      duration:
   486→        form.contentType === "video" ? form.duration.trim() || undefined : undefined,
   487→      fileSize: form.fileSize || undefined,
   488→      createdBy: adminEmail,
   489→    };
   490→
   491→    setSubmitting(true);
   492→    try {
   493→      if (editingId) {
   494→        const result = updateContent(editingId, payload);
   495→        if (result.success) {
   496→          toast({
   497→            title: "Konten diperbarui",
   498→            description: result.message,
   499→          });
   500→          handleCloseDialog();
   501→        } else {
   502→          toast({
   503→            title: "Gagal memperbarui",
   504→            description: result.message,
   505→            variant: "destructive",
   506→          });
   507→        }
   508→      } else {
   509→        const result = addContent(payload);
   510→        if (result.success) {
   511→          toast({
   512→            title: "Konten ditambahkan",
   513→            description: result.message,
   514→          });
   515→          handleCloseDialog();
   516→        } else {
   517→          toast({
   518→            title: "Gagal menambahkan",
   519→            description: result.message,
   520→            variant: "destructive",
   521→          });
   522→        }
   523→      }
   524→    } finally {
   525→      setSubmitting(false);
   526→    }
   527→  };
   528→
   529→  const handleConfirmDelete = () => {
   530→    if (!deleteTargetId) return;
   531→    const result = deleteContent(deleteTargetId);
   532→    if (result.success) {
   533→      toast({ title: "Konten dihapus", description: result.message });
   534→    } else {
   535→      toast({
   536→        title: "Gagal menghapus",
   537→        description: result.message,
   538→        variant: "destructive",
   539→      });
   540→    }
   541→    setDeleteTargetId(null);
   542→  };
   543→
   544→  const handleToggleActive = (id: string) => {
   545→    toggleActive(id);
   546→    toast({
   547→      title: "Status diperbarui",
   548→      description: "Status aktif konten telah diubah.",
   549→    });
   550→  };
   551→
   552→  const handleMoveUp = (content: EducationalContent) => {
   553→    const idx = filteredContents.findIndex((c) => c.id === content.id);
   554→    if (idx <= 0) return;
   555→    const prev = filteredContents[idx - 1];
   556→    updateOrder(content.id, prev.displayOrder);
   557→    updateOrder(prev.id, content.displayOrder);
   558→  };
   559→
   560→  const handleMoveDown = (content: EducationalContent) => {
   561→    const idx = filteredContents.findIndex((c) => c.id === content.id);
   562→    if (idx < 0 || idx >= filteredContents.length - 1) return;
   563→    const next = filteredContents[idx + 1];
   564→    updateOrder(content.id, next.displayOrder);
   565→    updateOrder(next.id, content.displayOrder);
   566→  };
   567→
   568→  // =====================================================
   569→  // RENDER GUARDS
   570→  // =====================================================
   571→
   572→  if (!mounted) {
   573→    return (
   574→      <div className="min-h-screen flex items-center justify-center bg-gradient-to-br from-gray-50 to-green-50/30">
   575→        <div className="h-8 w-8 rounded-full border-2 border-green-600 border-t-transparent animate-spin" />
   576→      </div>
   577→    );
   578→  }
   579→
   580→  if (!isAdmin) {
   581→    return (
   582→      <div className="min-h-screen flex items-center justify-center bg-gradient-to-br from-gray-50 to-green-50/30">
   583→        <div className="text-center px-6">
   584→          <div className="h-14 w-14 rounded-2xl bg-green-100 flex items-center justify-center mx-auto mb-3">
   585→            <LayoutGrid className="h-7 w-7 text-green-600" />
   586→          </div>
   587→          <p className="text-sm font-medium text-gray-700 mb-1">
   588→            Akses terbatas
   589→          </p>
   590→          <p className="text-xs text-gray-500">
   591→            Mengalihkan ke halaman login admin...
   592→          </p>
   593→        </div>
   594→      </div>
   595→    );
   596→  }
   597→
   598→  // =====================================================
   599→  // MAIN RENDER
   600→  // =====================================================
   601→
   602→  return (
   603→    <div className="animate-fade-in min-h-screen bg-gradient-to-br from-gray-50 to-green-50/30">
   604→      <div className="mx-auto max-w-6xl px-4 sm:px-6 lg:px-8 py-8 lg:py-12">
   605→        {/* Header */}
   606→        <div className="mb-6 flex flex-col sm:flex-row sm:items-center sm:justify-between gap-3">
   607→          <div>
   608→            <Badge
   609→              variant="secondary"
   610→              className="mb-1.5 bg-green-50 text-green-700 border-green-200 rounded-full"
   611→            >
   612→              <Pencil className="h-3 w-3 mr-1" />
   613→              Admin Panel
   614→            </Badge>
   615→            <h1 className="font-heading text-2xl sm:text-3xl font-extrabold text-gray-900">
   616→              Edit Website
   617→            </h1>
   618→            <p className="text-sm text-gray-600 mt-1">
   619→              Kelola konten edukasi website
   620→            </p>
   621→          </div>
   622→          <Button
   623→            onClick={handleOpenAdd}
   624→            className="rounded-full bg-green-600 hover:bg-green-700 text-white shadow-md self-start sm:self-auto"
   625→          >
   626→            <Plus className="h-4 w-4 mr-1.5" />
   627→            Tambah Konten
   628→          </Button>
   629→        </div>
   630→
   631→        {/* Search */}
   632→        <Card className="border-0 shadow-md rounded-2xl overflow-hidden mb-4">
   633→          <CardContent className="pt-4 pb-4">
   634→            <div className="relative">
